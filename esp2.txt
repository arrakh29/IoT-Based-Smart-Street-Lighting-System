#include <ThingSpeak.h>
#include <UniversalTelegramBot.h>
#include <WiFiClientSecure.h>

#define WIFI_SSID "inpo"
#define WIFI_PASSWORD "miskinbanget"

// Ganti dengan detail Firebase Anda
#define FIREBASE_HOST "https://projek-94e51-default-rtdb.asia-southeast1.firebasedatabase.app/"
#define FIREBASE_AUTH "AIzaSyCOsCde0MA2SaulZiZ12IVjig1qL7geXgo"

// Ganti dengan API key ThingSpeak
#define THINGSPEAK_API_KEY "Q8ALEEZC9567A69U"
unsigned long myChannelNumber = 1;

// Ganti dengan token bot Telegram
#define TELEGRAM_BOT_TOKEN "7456245298:AAFMqwZ1dAS_qJWC4EV1qwW8TMCUrTjhq80"
#define CHAT_ID "1807778211"

// Objek Firebase dan ThingSpeak
FirebaseData firebaseData;
FirebaseAuth auth;
FirebaseConfig config;
WiFiClient client;
WiFiClientSecure secured_client;
UniversalTelegramBot bot(TELEGRAM_BOT_TOKEN, secured_client);

void setup() {
  Serial.begin(115200);

  // Menghubungkan ke WiFi
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  Serial.print("Connecting to Wi-Fi");
  while (WiFi.status() != WL_CONNECTED) {
    Serial.print(".");
    delay(1000);
  }
  Serial.println();
  Serial.print("Connected to Wi-Fi, IP address: ");
  Serial.println(WiFi.localIP());

  // Konfigurasi Firebase
  config.host = FIREBASE_HOST;
  config.signer.tokens.legacy_token = FIREBASE_AUTH;

  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);
  Serial.println("Connected to Firebase");

  // Menghubungkan ke ThingSpeak
  ThingSpeak.begin(client);
  secured_client.setInsecure();
}

void loop() {
  // Mengambil data dari Firebase
  if (Firebase.getInt(firebaseData, "/carDetected")) {
    int carDetected = firebaseData.intData();
    Serial.print("Car Detected: ");
    Serial.println(carDetected);

    // Mengirim data ke ThingSpeak
    int responseCode = ThingSpeak.writeField(myChannelNumber, 1, carDetected, THINGSPEAK_API_KEY);
    if (responseCode == 200) {
      Serial.println("Data sent to ThingSpeak successfully");
    } else {
      Serial.println("Error sending data to ThingSpeak: " + String(responseCode));
    }

    // Mengirim pesan ke Telegram
    String message = "Car Detected: " + String(carDetected);
    if (bot.sendMessage(CHAT_ID, message, "")) {
      Serial.println("Message sent to Telegram successfully");
    } else {
      Serial.println("Failed to send message to Telegram");
    }
  } else {
    Serial.print("Failed to get data: ");
    Serial.println(firebaseData.errorReason());
  }

  delay(10000); // Delay untuk mengurangi frekuensi pembacaan
}
